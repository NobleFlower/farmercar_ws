cmake_minimum_required(VERSION 3.0.2)
project(apriltag_servo)
find_package(catkin REQUIRED COMPONENTS rospy geometry_msgs std_msgs apriltag_ros tf2_ros tf2_geometry_msgs)
catkin_python_setup()
catkin_package()
catkin_install_python(PROGRAMS scripts/apriltag_follow_preview.py DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_nosetests(test/test_control.py)
  find_package(rostest REQUIRED)
  add_rostest(test/faults.test)
endif()

install(FILES README.md DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
